local function sendAdminCommand(target, cmd)
    if adminRemote and adminRemote:IsA("RemoteEvent") then
        pcall(function()
            adminRemote:FireServer(target, cmd)
        end)
        return true
    elseif adminRemote and adminRemote:IsA("RemoteFunction") then
        pcall(function()
            adminRemote:InvokeServer(target, cmd)
        end)
        return true
    end
    return false
end

local WALL_BOUNDS = {
    {xMin = -370, xMax = -285, yMin = -15, yMax = 150, zMin = 25, zMax = 35},
}
local wallPlayerLastPos = {}
local wallCooldowns = {}

local function scanForSteal()
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not playerGui then return false end
    for _, v in ipairs(playerGui:GetDescendants()) do
        if (v:IsA("TextLabel") or v:IsA("TextButton")) and v.Visible then
            local text = v.Text or ""
            if text:find("Someone is stealing your") then
                return true
            end
        end
    end
    return false
end

local function checkWallCrossing(player)
    if not autoBalloonEnabled or not isStealingActive then return end
    if not player or player == LocalPlayer then return end
    local char = player.Character
    if not char then return end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    local uid = player.UserId
    local currentPos = hrp.Position
    local lastPos = wallPlayerLastPos[uid]
    if not lastPos then
        wallPlayerLastPos[uid] = currentPos
        return
    end
    for _, wall in ipairs(WALL_BOUNDS) do
        local lastIn = (lastPos.X >= wall.xMin and lastPos.X <= wall.xMax and
                        lastPos.Y >= wall.yMin and lastPos.Y <= wall.yMax and
                        lastPos.Z >= wall.zMin and lastPos.Z <= wall.zMax)
        local currIn = (currentPos.X >= wall.xMin and currentPos.X <= wall.xMax and
                        currentPos.Y >= wall.yMin and currentPos.Y <= wall.yMax and
                        currentPos.Z >= wall.zMin and currentPos.Z <= wall.zMax)
        if lastIn ~= currIn then
            if not wallCooldowns[uid] or tick() - wallCooldowns[uid] > 2 then
                wallCooldowns[uid] = tick()
                fireAdminBalloon(player)
            end
        end
    end
    wallPlayerLastPos[uid] = currentPos
end

end)

local function applyTinyAndRocket(target)
    if not target then return end
    if sendAdminCommand(target, "tiny") then
        task.wait(randomDelay(0.05, 0.15))
        sendAdminCommand(target, "rocket")
    else
        local channel = TextChatService and TextChatService.TextChannels:FindFirstChild("RBXGeneral")
        if channel then
            local n = target.Name
            task.spawn(function()
                pcall(channel.SendAsync, channel, ";tiny " .. n)
                task.wait(randomDelay(0.05, 0.15))
                pcall(channel.SendAsync, channel, ";rocket " .. n)
            end)
        else
            fireAdminCommand(target, "tiny")
            task.wait(randomDelay(0.05, 0.15))
            fireAdminCommand(target, "rocket")
        end
    end
end

local function getClosestPlayer()
    local localChar = LocalPlayer.Character
    if not localChar or not localChar:FindFirstChild("HumanoidRootPart") then return nil end
    local pos = localChar.HumanoidRootPart.Position
    local closest, closestDist = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            local char = p.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                local d = (char.HumanoidRootPart.Position - pos).Magnitude
                if d < closestDist then
                    closestDist = d
                    closest = p
                end
            end
        end
    end
    return closest
end

local function findTool(name)
    if not Character then return nil end
    for _, tool in ipairs(Character:GetChildren()) do
        if tool:IsA("Tool") and tool.Name:lower():find(name:lower()) then return tool end
    end
    for _, tool in ipairs(LocalPlayer.Backpack:GetChildren()) do
        if tool:IsA("Tool") and tool.Name:lower():find(name:lower()) then return tool end
    end
    return nil
end

local function isValidStealPrompt(prompt)
    if not prompt or not prompt.Parent or not prompt.Enabled then return false end
    local state = prompt:GetAttribute("State")
    local actionText = prompt.ActionText
    if state == "Steal" or state == "Grab" or actionText == "Steal" or actionText == "Grab" then return true end
    return false
end

local function executeSteal(prompt)
    if isStealing or not prompt or not prompt.Parent then return end
    isStealing = true
    prompt:HoldButton()
    task.wait(stealDelay + randomDelay(-0.1, 0.1))
    prompt:Trigger()
    task.wait(randomDelay(0.02, 0.06))
    isStealing = false
end

local STOP_DIST = 5 + math.random() * 0.5
local SLOW_DIST = 20 + math.random() * 2

local function startTripToPetSlot(prompt, slotNumber)
    local config = SlotsConfig[slotNumber] or SlotsConfig[1]
    local targetPositions = config.Positions
    local needJump = config.NeedJump == true
    if slotNumber >= 19 and slotNumber <= 27 then needJump = true end

    if currentMovement then pcall(currentMovement.Disconnect); currentMovement = nil end
    if not Root or not Humanoid then return end

    autoStealEnabled = true
    local Speed = 200 + math.random() * 20
    local grabStartDistance = 60 + math.random() * 5
    local grabStarted = false

    local carpet = findTool("flying carpet")
    if carpet then
        Humanoid:UnequipTools()
        task.wait(randomDelay(0.02, 0.06))
        Humanoid:EquipTool(carpet)
    end

    if Root:FindFirstChild("LinearVelocity") then Root.LinearVelocity:Destroy() end
    if Root:FindFirstChild("Attachment") then Root.Attachment:Destroy() end

    if antiSwapEnabled then
        task.spawn(function()
            task.wait(randomDelay(0.1, 0.25))
            applyTinyAndRocket(getClosestPlayer())
        end)
    end

    local Attachment = Instance.new("Attachment")
    Attachment.Parent = Root
    local Velocity = Instance.new("LinearVelocity")
    Velocity.Attachment0 = Attachment
    Velocity.RelativeTo = Enum.ActuatorRelativeTo.World
    Velocity.MaxForce = math.huge
    Velocity.Parent = Root

    local currentPosIndex = 1
    local intermediatePauseActive = false

    currentMovement = RunService:BindToRenderStep("Movement", 150, function()
        if not Root or not Root.Parent or Humanoid.Health <= 0 then
            if currentMovement then RunService:UnbindFromRenderStep("Movement"); currentMovement = nil end
            return
        end
        if intermediatePauseActive then
            Velocity.VectorVelocity = Vector3.zero
            return
        end

        local TargetPosition = targetPositions[currentPosIndex]
        if not TargetPosition then return end

        local rootPos = Root.Position
        local dir = Vector3.new(TargetPosition.X - rootPos.X, 0, TargetPosition.Z - rootPos.Z)
        local dist = dir.Magnitude

        if currentPosIndex == 1 and dist <= grabStartDistance and not grabStarted then
            grabStarted = true
            task.spawn(function() executeSteal(prompt) end)
        end

        local speedMult = 1
        if dist < SLOW_DIST then speedMult = math.max(0.15, dist / SLOW_DIST) * jitter(1, 0.05) end
        if dist <= STOP_DIST then
            if currentPosIndex < #targetPositions then
                intermediatePauseActive = true
                Velocity.VectorVelocity = Vector3.zero
                Root.AssemblyLinearVelocity = Vector3.zero
                task.spawn(function()
                    task.wait(randomDelay(0.1, 0.25))
                    currentPosIndex = currentPosIndex + 1
                    intermediatePauseActive = false
                end)
                return
            end

            Velocity.VectorVelocity = Vector3.zero
            Root.AssemblyLinearVelocity = Vector3.zero
            Velocity:Destroy()
            Attachment:Destroy()
            Root.CFrame = CFrame.new(TargetPosition)
            if currentMovement then RunService:UnbindFromRenderStep("Movement"); currentMovement = nil end

            task.wait(randomDelay(0.1, 0.2))
            Camera.CameraType = Enum.CameraType.Scriptable
            Camera.CFrame = CFrame.new(Root.Position + config.CamOffset) * CFrame.Angles(unpack(config.CamAngles))
            Humanoid:UnequipTools()
            task.wait(randomDelay(0.04, 0.1))

            if needJump then
                Root.AssemblyLinearVelocity = Vector3.new(0, 55 + math.random() * 10, 0)
                task.wait(randomDelay(0.06, 0.12))
            end

            local flash = findTool("flash")
            if flash then
                Humanoid:EquipTool(flash)
                task.wait(randomDelay(0.06, 0.12))
                flash:Activate()
            end

            task.wait(randomDelay(0.08, 0.15))
            if ragdollBypassEnabled then
                local giant = findTool("giant potion")
                if giant then
                    Humanoid:EquipTool(giant)
                    task.wait(randomDelay(0.06, 0.12))
                    giant:Activate()
                    task.wait(randomDelay(0.02, 0.05))
                    Humanoid:UnequipTools()
                end
                fireSelfCommand("ragdoll")
            end

            Camera.CameraType = Enum.CameraType.Custom
            task.spawn(function()
                task.wait(randomDelay(0.8, 1.2))
                autoStealEnabled = false
            end)
            return
        end

        Velocity.VectorVelocity = Vector3.new(dir.Unit.X * Speed * speedMult, 0, dir.Unit.Z * Speed * speedMult)
    end)
end

local autoGrabConn = nil

local function findNearestAutoGrabPrompt()
    local c = LocalPlayer.Character
    local hrp = c and c:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local plots = Workspace:FindFirstChild("Plots")
    if not plots then return nil end
    local nearest, dist = nil, math.huge
    for _, plot in ipairs(plots:GetChildren()) do
        if not isMyPlot(plot) then
            local pods = plot:FindFirstChild("AnimalPodiums")
            if pods then
                for _, pod in ipairs(pods:GetChildren()) do
                    local base = pod:FindFirstChild("Base")
                    if base then
                        local sp = base:FindFirstChild("Spawn")
                        if sp then
                            local d = (sp.Position - hrp.Position).Magnitude
                            if d <= STEAL_RADIUS and d < dist then
                                local att = sp:FindFirstChild("PromptAttachment")
                                if att then
                                    for _, p in ipairs(att:GetChildren()) do
                                        if p:IsA("ProximityPrompt") and isValidStealPrompt(p) then
                                            nearest, dist = p, d
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return nearest
end

local function executeAutoGrab(prompt)
    if autoGrabStealing then return end
    autoGrabStealing = true
    local startTime = tick()
    prompt:HoldButton()
    while tick() - startTime < STEAL_DURATION do
        task.wait()
    end
    prompt:Trigger()
    task.wait(randomDelay(0.02, 0.08))
    autoGrabStealing = false
end

local function startAutoGrab()
    if autoGrabConn then return end
    autoGrabConn = RunService:BindToRenderStep("AutoGrab", 100, function()
        if not autoGrabEnabled or autoGrabStealing or autoStealEnabled then return end
        local prompt = findNearestAutoGrabPrompt()
        if prompt then executeAutoGrab(prompt) end
    end)
end

local function stopAutoGrab()
    if autoGrabConn then RunService:UnbindFromRenderStep("AutoGrab"); autoGrabConn = nil end
    autoGrabStealing = false
    AutoGrabData = {}
end

local activeHighlights = {}
local activeBillboards = {}

local function removeESP(targetPlayer)
    local char = targetPlayer.Character
    if not char then return end
    local hl = char:FindFirstChild("AdminESP")
    if hl then hl:Destroy() end
    local head = char:FindFirstChild("Head")
    if head then
        local tag = head:FindFirstChild("AdminTag")
        if tag then tag:Destroy() end
    end
    activeHighlights[targetPlayer] = nil
    activeBillboards[targetPlayer] = nil
end

local function createESP(targetPlayer, isAdmin)
    local char = targetPlayer.Character
    if not char or targetPlayer == LocalPlayer then return end
    removeESP(targetPlayer)

    local hl = Instance.new("Highlight")
    hl.Name = "AdminESP"
    hl.FillColor = isAdmin and Color3.fromRGB(80,80,80) or Color3.fromRGB(50,50,50)
    hl.OutlineColor = isAdmin and Color3.fromRGB(100,100,100) or Color3.fromRGB(40,40,40)
    hl.FillTransparency = 0.5
    hl.OutlineTransparency = 0
    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
    hl.Parent = char
    activeHighlights[targetPlayer] = hl

    local head = char:FindFirstChild("Head")
    if head then
        local billboard = Instance.new("BillboardGui")
        billboard.Name = "AdminTag"
        billboard.AlwaysOnTop = true
        billboard.Size = UDim2.new(0, 110, 0, 38)
        billboard.StudsOffset = Vector3.new(0, 2.5, 0)
        billboard.Parent = head
        activeBillboards[targetPlayer] = billboard

        local frame = Instance.new("Frame")
        frame.Size = UDim2.new(1,0,1,0)
        frame.BackgroundTransparency = 0.1
        frame.BackgroundColor3 = Color3.fromRGB(20,20,25)
        frame.Parent = billboard
        Instance.new("UICorner", frame).CornerRadius = UDim.new(0,10)
        local label = Instance.new("TextLabel", frame)
        label.Size = UDim2.new(1,0,1,0)
        label.BackgroundTransparency = 1
        label.Text = isAdmin and "AP" or "No AP"
        label.TextColor3 = isAdmin and Color3.fromRGB(200,200,200) or Color3.fromRGB(120,120,130)
        label.TextStrokeTransparency = 0.3
        label.Font = Enum.Font.GothamBold
        label.TextSize = 15
        label.TextXAlignment = Enum.TextXAlignment.Center
        label.TextYAlignment = Enum.TextYAlignment.Center
    end
end

local function checkPlayerESP(targetPlayer)
    if not apEspEnabled or targetPlayer == LocalPlayer then return end
    if targetPlayer.Character then
        local isAdmin = targetPlayer:GetAttribute("AdminCommands") == true
        createESP(targetPlayer, isAdmin)
    else
        removeESP(targetPlayer)
    end
end

local function setupPlayerESP(targetPlayer)
    if targetPlayer == LocalPlayer then return end
    task.wait(randomDelay(0.3, 0.8))
    checkPlayerESP(targetPlayer)
    targetPlayer.CharacterAdded:Connect(function()
        task.wait(randomDelay(0.3, 0.8))
        checkPlayerESP(targetPlayer)
    end)
    targetPlayer:GetAttributeChangedSignal("AdminCommands"):Connect(function()
        checkPlayerESP(targetPlayer)
    end)
end

local function enableAPESP()
    for _, p in ipairs(Players:GetPlayers()) do setupPlayerESP(p) end
    apEspEnabled = true
end

local function disableAPESP()
    apEspEnabled = false
    for _, p in ipairs(Players:GetPlayers()) do removeESP(p) end
    for k in pairs(activeHighlights) do activeHighlights[k] = nil end
    for k in pairs(activeBillboards) do activeBillboards[k] = nil end
end

Players.ChildAdded:Connect(function(child)
    if child:IsA("Player") then
        task.wait(randomDelay(0.5, 1.5))
        if apEspEnabled then setupPlayerESP(child) end
    end
end)

local function getDevice()
    local screen = workspace.CurrentCamera.ViewportSize
    local w, h = screen.X, screen.Y
    local isMobile = UserInputService.TouchEnabled and not UserInputService.KeyboardEnabled
    if isMobile then
        if w >= 900 or h >= 900 then return "ipad" end
        return "mobile"
    end
    return "pc"
end

local function toggleRow(title, desc, defaultOn, onToggle)
    sectionOrder = sectionOrder + 1
    local row = Instance.new("Frame", ScrollingFrame2)
    row.Size = UDim2.new(1,0,0,36)
    row.BackgroundColor3 = C.card
    row.BorderSizePixel = 0
    row.ZIndex = 4
    row.LayoutOrder = sectionOrder
    Instance.new("UICorner", row)
    local s = Instance.new("UIStroke", row); s.Color = C.stroke
    local accentBar = Instance.new("Frame", row)
    accentBar.Size = UDim2.new(0,3,1,-10)
    accentBar.Position = UDim2.new(0,0,0,5)
    accentBar.BackgroundColor3 = C.accent
    accentBar.BorderSizePixel = 0
    accentBar.ZIndex = 5
    Instance.new("UICorner", accentBar).CornerRadius = UDim.new(0,2)
    local t1 = Instance.new("TextLabel", row)
    t1.Size = UDim2.new(1,-52,1,0)
    t1.Position = UDim2.new(0,12,0,0)
    t1.BackgroundTransparency = 1
    t1.ZIndex = 5
    t1.Text = title
    t1.TextColor3 = C.textBright
    t1.TextSize = 13
    t1.Font = Enum.Font.GothamMedium
    t1.TextXAlignment = Enum.TextXAlignment.Left

    local track = Instance.new("Frame", row)
    track.Size = UDim2.new(0,36,0,20)
    track.Position = UDim2.new(1,-42,0.5,-10)
    track.BackgroundColor3 = C.accent
    track.BorderSizePixel = 0
    track.ZIndex = 6
    Instance.new("UICorner", track).CornerRadius = UDim.new(0,10)
    local ts = Instance.new("UIStroke", track); ts.Color = C.accent
    local knob = Instance.new("Frame", track)
    knob.Size = UDim2.new(0,14,0,14)
    knob.Position = UDim2.new(0,19,0.5,-7)
    knob.BackgroundColor3 = C.knobOn
    knob.BorderSizePixel = 0
    knob.ZIndex = 7
    Instance.new("UICorner", knob).CornerRadius = UDim.new(0,7)
    local hit = Instance.new("TextButton", row)
    hit.Size = UDim2.new(1,0,1,0)
    hit.BackgroundTransparency = 1
    hit.ZIndex = 8
    hit.Text = ""

    local ref = { on = defaultOn ~= false, track = track, stroke = ts, knob = knob }
    local function render(animate)
        local info = TweenInfo.new(animate and 0.16 or 0, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        if ref.on then
            TweenService:Create(track, info, { BackgroundColor3 = C.accent }):Play()
            TweenService:Create(ts, info, { Color = C.accent }):Play()
            TweenService:Create(knob, info, { Position = UDim2.new(0,19,0.5,-7), BackgroundColor3 = C.knobOn }):Play()
        else
            TweenService:Create(track, info, { BackgroundColor3 = C.trackOff }):Play()
            TweenService:Create(ts, info, { Color = C.stroke }):Play()
            TweenService:Create(knob, info, { Position = UDim2.new(0,3,0.5,-7), BackgroundColor3 = C.knobOff }):Play()
        end
    end
    render(false)
    hit.MouseButton1Click:Connect(function()
        ref.on = not ref.on
        render(true)
        if onToggle then pcall(onToggle, ref.on) end
        saveSetting(title, ref.on)
    end)
    table.insert(toggleRefs, ref)
    return row, ref
end

local function syncBorder()
    BorderFrame.Position = Win.Position
end
do
    local dragging, dragStart, startPos
    local function begin(input)
        dragging = true
        dragStart = input.Position
        startPos = Win.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
    Frame3.Active = true
    Frame3.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            begin(input)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            Win.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
            syncBorder()
        end
    end)
end

local function hookButton(btn, normal, hover)
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundColor3 = hover }):Play()
    end)
    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundColor3 = normal }):Play()
    end)
    btn.MouseButton1Down:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.06), { BackgroundColor3 = C.deepBlue }):Play()
    end)
    btn.MouseButton1Up:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.1), { BackgroundColor3 = hover }):Play()
    end)
end

local flashBlinkActive = false
local BLINK_HI = Color3.fromRGB(200,220,255)
local BLINK_LO = Color3.fromRGB(60,70,90)
local BLINK_T = TweenInfo.new(0.3, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
local function startFlashBlink()
    if flashBlinkActive then return end
    flashBlinkActive = true
    task.spawn(function()
        while flashBlinkActive do
            TweenService:Create(FLASHTP, BLINK_T, { BackgroundColor3 = BLINK_HI }):Play()
            TweenService:Create(flashAccent, BLINK_T, { BackgroundColor3 = BLINK_HI }):Play()
            TweenService:Create(Frame13, BLINK_T, { BackgroundColor3 = BLINK_HI, BackgroundTransparency = 0 }):Play()
            task.wait(0.3)
            if not flashBlinkActive then break end
            TweenService:Create(FLASHTP, BLINK_T, { BackgroundColor3 = BLINK_LO }):Play()
            TweenService:Create(flashAccent, BLINK_T, { BackgroundColor3 = BLINK_LO }):Play()
            TweenService:Create(Frame13, BLINK_T, { BackgroundColor3 = BLINK_LO }):Play()
            task.wait(0.3)
        end
    end)
end
local function stopFlashBlink()
    flashBlinkActive = false
    TweenService:Create(FLASHTP, TweenInfo.new(0.2), { BackgroundColor3 = C.card }):Play()
    TweenService:Create(flashAccent, TweenInfo.new(0.2), { BackgroundColor3 = C.stroke }):Play()
    TweenService:Create(Frame13, TweenInfo.new(0.2), { BackgroundColor3 = C.accent, BackgroundTransparency = 0 }):Play()
end

local function purge()
    stopAutoGrab()
    disableAPESP()
    if charAddedConn then charAddedConn:Disconnect() end
    if currentMovement then RunService:UnbindFromRenderStep("Movement") end
    RunService:UnbindFromRenderStep("BalloonCheck")
    RunService:UnbindFromRenderStep("AutoGrab")
    LEVITHON_GUI:Destroy()
    _G.Levithon_Script_Purge = nil
end
